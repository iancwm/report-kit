% publications put on their publication-details page.
\newcommand{\RKBriefMetaIfSet}[2]{%
  \if\relax\detokenize\expandafter{#2}\relax\else\briefmeta{#1}{#2}\fi
}
\newcommand{\RKFrontMatter}{%
  % Sections flow continuously in a brief; long-form page-per-section
  % pagination would spread two pages of argument over eight.
  \RKSectionOpensPagefalse
  \ifRKPubHasCover
    \clearpage\thispagestyle{empty}\noindent
    \makebox[\textwidth][c]{\includegraphics[page=1,width=\textwidth,height=\textheight,keepaspectratio]{\RKPubCoverPath}}\clearpage
  \fi
  \begin{briefheader}{\RKPubLeftHeader}{\RKPubTitle}
    \RKPubSubtitle
    \RKBriefMetaIfSet{Prepared by}{\RKPubAuthor}
    \RKBriefMetaIfSet{Date}{\RKPubDate}
    \RKBriefMetaIfSet{Version}{\RKPubVersion}
    \RKBriefMetaIfSet{Classification}{\RKPubClassification}
  \end{briefheader}

  \fi
